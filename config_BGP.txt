========ASR-CORE1========
en
conf t
host ASR-CORE1

int g0/0
ip add 192.168.1.20 255.255.255.0
no sh
bfd interval 300 min_rx 300 multipl 3

int g0/1
no ip add
no sh

int gi0/1.200
description GATEWAY-BASE-CLIENT-1
encapsulation dot1Q 200
ip add 198.51.100.7 255.255.255.240

vrrp 3 ip 198.51.100.1
vrrp 3 prio 100
no sh

int gi0/1.400
description GATEWAY-BASE-CLIENT-2
encapsulation dot1Q 400
ip add 203.0.113.4 255.255.255.240

vrrp 1 ip 203.0.113.1
vrrp 1 prio 110
vrrp 1 preempt
no sh

int gi0/1.900
description RESEAU-INTERNE-OPERATEUR
encapsulation dot1Q 900
ip add 10.1.1.40 255.255.0.0
no sh
exit

route-map BASE-CLIENT1 permit 10
set communi 65001:100 additi

route-map BASE-CLIENT2 permit 10
set communi 65001:200 additi

route-map PREPEND-UPSTREAM1 permit 10
set as-path prepend 65001 65001 65001

exit

ip access-list stand ACL-VIP
permit host 198.51.100.8
permit host 198.51.100.9
permit host 198.51.100.10
permit host 198.51.100.11
permit host 198.51.100.12
permit host 198.51.100.13
permit host 198.51.100.14
permit host 203.0.113.5
permit host 203.0.113.3

exit

class-map match-all CLASS-VIP
match acces-group name ACL-VIP

exit

ip prefix-list UPSTREAM1-PREFERED seq 10 permit 3.3.3.3/32

route-map RM-UPSTREAM1-IN permit 10
match ip add prefix-list UPSTREAM1-PREFERED
set local-preference 300
exit

route-map RM-UPSTREAM1-IN permit 20
exit

router bgp 65001
neigh 192.168.1.21 remote-as 65002
neigh 192.168.1.21 route-map PREPEND-UPSTREAM1 out
neigh 192.168.1.21 route-map RM-UPSTREAM1-IN in
neigh 192.168.1.21 timers 10 30
neigh 192.168.1.21 fall-over bfd
neigh 10.1.1.42 remote-as 65001
neigh 10.1.1.42 description IBGP-VERS-ASR-CORE-2
neigh 10.1.1.42 update-source gi0/1.900
neigh 10.1.1.42 next-hop-self

netw 198.51.100.0 mask 255.255.255.0
netw 203.0.113.0 mask 255.255.255.0
netw 198.51.100.0 mask 255.255.255.0 route-map BASE-CLIENT1
netw 203.0.113.0 mask 255.255.255.0 route-map BASE-CLIENT2

ex

track 1 int g0/1 line-protocol
delay down 10 up 10

ip route 198.51.100.0 255.255.255.0 Null0 track 1
ip route 203.0.113.0 255.255.255.0 Null0 track 1

policy-map QOS
class CLASS-VIP
prio 46000
class class-default
bandwidth 70000
exit

policy-map LIMIT-UPSTREAM1
class class-default
shape average 155000000
service-policy QOS

int g0/0
service-policy output LIMIT-UPSTREAM1
exit

end
clear ip bgp * soft

do wr mem
